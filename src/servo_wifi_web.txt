#include <WiFi.h>

// Test servo me faqe web + komanda nga Serial Monitor.
// Lidhu me WiFi "ServoTest" dhe hap http://192.168.4.1
// Serial Monitor:  start | stop | 0..180

// ---------- CILËSIMET ----------

const char* AP_PASS = "servo12345";   // të paktën 8 shkronja
const int SERVO_PIN = 27;
// --------------------------------

Servo servo;
WebServer server(80);

bool sweeping = false;          // a po lëviz vetë apo jo
int currentAngle = 90;
const int steps[] = {0, 90, 180, 90};
int stepIndex = 0;
unsigned long lastMoveMs = 0;

// Faqja web (HTML + CSS + JavaScript), ruhet brenda ESP32

void moveTo(int angle) {
  currentAngle = angle;
  Serial.printf("Po shkoj ne %d gradë\n", angle);
  servo.write(angle);
}

void sendState() {
  String json = String("{\"angle\":") + currentAngle +
                ",\"sweeping\":" + (sweeping ? "true" : "false") + "}";
  server.send(200, "application/json", json);
}

void handleCommand(String cmd) {
  cmd.trim();
  cmd.toLowerCase();
  if (cmd == "start") {
    sweeping = true;
    Serial.println(">>> START");
  } else if (cmd == "stop") {
    sweeping = false;
    Serial.println(">>> STOP");
  } else if (cmd.length() > 0 && isDigit(cmd[0])) {
    int angle = cmd.toInt();
    if (angle >= 0 && angle <= 180) {
      sweeping = false;   // kur jep kënd me dorë, ndalon lëvizja automatike
      moveTo(angle);
    } else {
      Serial.println("Këndi duhet të jetë nga 0 deri 180");
    }
  } else if (cmd.length() > 0) {
    Serial.println("Komandat: start, stop, ose një numër 0-180");
  }
}

void setup() {
  Serial.begin(115200);
  servo.setPeriodHertz(50);              // servot standarde punojnë me 50 Hz
  servo.attach(SERVO_PIN, 500, 2400);    // gjatësia e pulsit për 0° dhe 180°
  moveTo(90);                            // fillon në mes dhe pret komandë

  WiFi.mode(WIFI_AP);
  WiFi.softAP(AP_SSID, AP_PASS);
  Serial.print("WiFi 'ServoTest' gati. Hap ne browser: http://");
  Serial.println(WiFi.softAPIP());

  server.on("/", []() { server.send_P(200, "text/html", PAGE); });
  server.on("/state", sendState);
  server.on("/start", []() { handleCommand("start"); sendState(); });
  server.on("/stop",  []() { handleCommand("stop");  sendState(); });
  server.on("/set", []() {
    if (server.hasArg("angle")) handleCommand(server.arg("angle"));
    sendState();
  });
  server.begin();

  Serial.println("Gati. Shkruaj: start, stop, ose një numër 0-180");
}

void loop() {
  server.handleClient();

  if (Serial.available()) {
    handleCommand(Serial.readStringUntil('\n'));
  }

  if (sweeping && millis() - lastMoveMs > 1500) {
    lastMoveMs = millis();
    moveTo(steps[stepIndex]);
    stepIndex = (stepIndex + 1) % 4;
  }
}
